46function codeToChar( code )
47{
48    // Unicode character from an integer between 0 and 65535 (0xFFFF). Numbers greater than 0xFFFF are truncated.
49    return String.fromCharCode( code ) ;
50}

52function charToCode( ch )
53{
54    // Returns an integer between 0 and 65535 representing the UTF-16 code unit at string index 0.
55    return ch.charCodeAt( 0 ) ;
56}
