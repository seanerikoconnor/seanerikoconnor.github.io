  27/*
  28 * HTML elements are two main types.
  29 *
  30 *   Block elements:  Each one starts on a new line.
  31 *
  32 *     +--------+
  33 *     | block1 |
  34 *     +--------+  <-- vertical margins collapse where blocks overlap.
  35 *     +--------+  <-- 
  36 *     | block2 |
  37 *     +--------+
  38 *
  39 *   Inline elements:  Go with the flow, left to right for English, 
  40 *   right to left for Hebrew.
  41 *
  42 *     +---------+   +---------+
  43 *     | inline1 |   | inline2 |
  44 *     +---------+   +---------+
  45 *  
  46 * Floating elements:  Text flows around this element which is floated left.
  47 *
  48 *     text text text text text text text text
  49 *     +---------+ text text text text text 
  50 *     | element | text text text text text 
  51 *     +---------+ text text text text 
  52 *     Hier stehe Ich, Ich kann nicht anders!     <----- <p style="clear: left;"> Hier stehe Ich, Ich kann nicht anders!
  53 *                                                       But specifying text with "clear" makes the text start BELOW any element 
  54 *                                                       which is floated left above it.
  55 * CSS Box Model
  56 *
  57 *     +------------------------------------+
  58 *     |  margin                            |
  59 *     |   +-----------border------------+  |
  60 *     |   |  padding                    |  |
  61 *     |   |     +-----------------+     |  |
  62 *     |   |     |C O N T E  N   T |     |  |
  63 *     |   |     |+..........+ C O |     |  |
  64 *     |   |     |. floating . N T |     |  | <------ Floating elements stay within the CONTENT.
  65 *     |   |     |. element  . E N |     |  |        
  66 *     |   |     |+..........+ E N |     |  |
  67 *     |   |     |T   C O N T E  N |     |  |
  68 *     |   |     |T C O N T E N  T |     |  |
  69 *     |   |     +-----------------+     |  |
  70 *     |   |                     padding |  |
  71 *     |   +-----------border------------+  |
  72 *     |                            margin  |
  73 *     +------------------------------------+
  74 *
  75 * CSS Nomenclature
  76 * 
  77 *  selector
  78 *   |
  79 * --+---                    ---+
  80 * em.wow                       |
  81 * {                            |
  82 *     font-style:  italic ;    |
  83 *     --------+------------    |
  84 *             |                |
  85 *          declaration #1      |
  86 *                              +---- rule
  87 *      property   value        |
  88 *         |         |          |
  89 *     ----+-----   -+--        |
  90 *     font-weight: bold ;      |
  91 * }   --------+--------        |
  92 *             |             ---+
  93 *        declaration #2
  94 * 
  95 * 
  96 *  Used in HTML as:
  97 *     <em class="wow">wowish talk talk</em>
  98 */

 358/*
 359 *   wrapper-aside-col contains the margin with the border image.
 360 *      -We make slightly larger than the img.border class to give a thin space on the right.
 361 *   wrapper-nav-col is the vertical navigation sidebar.  
 362 *      -Minimum is set wide enough so words in the navigation hyperlinks don't overflow to the right.  
 363 *      -Maximum is clamped so very wide page doesn't have big stretched out navigation hyperlinks.
 364 *   wrapper-content-col is most of the page, so it gets the most relative space.  
 365 *      -We use high fr number so that a very wide page shows more content and less navigation hyperlinks. 
 366 *   Recall fr is the fraction of the width AFTER subtracting fixed widths such as 1.5rem.
 367 *
 368 *   wrapper-*-row are set to auto because we want the default behavior of HTML flow;  we don't want to set the height of rows.
 369 */
 370div.wrapper
 371{
 372    display:                grid ;  /* Items in this container are grid type.  They are placed in 2D */
 373    grid-template-columns:  [wrapper-aside-col] 1.5rem [wrapper-nav-col] minmax( 9rem, 1fr ) [wrapper-content-col] 30fr [wrapper-end-col] ;
 374    grid-template-rows:     [wrapper-title-row] auto [wrapper-nav-row] auto [wrapper-article-row] auto [wrapper-footer-row] auto [wrapper-end-row] ;
 375}

1627 * ----------------------- Tooltips -----------------------------------------
1628 *
1629 *     Show tooltip contents (text only) when hovering a mouse or pen controlled cursor over the tooltip element.
1630 *     Mobile devices don't support hover, even with long press gestures.
1631 *
1632 *     Sample HTML:
1633 *         <span data-text="I am the tooltip contents." class="I'm a tooltip:  hover over me to see tooltip contents!">Tooltip text</span>
1634 */
1635@media (any-hover: hover)
1636{
1637    /*  Make the style visible as a tooltip. */
1638    span.tooltip
1639    {
1640        /* Since we don't specify any top/bottom/left/right property/values which would shift this element 
1641           relative TO ITSELF, this is the same as the default static positioning, i.e. normal document flow.
1642           Why do we need this?  Because the child class span.tooltip:before uses absolute positioning; 
1643           it needs this parent class to have either relative or absolute positioning for that to work. */
1644        position:      relative ;
1645
1646        /* Indicate the tooltip be a dashed line below its text. */
1647        border-bottom: 0.2rem dashed var(--art-gallery-blue-gray);
1648    }
1649
1650    /* Tooltip content.
1651     * This child class (:before) of the parent class (span.tooltip) is a pseudoclass which inserts content 
1652     * before the first item in the tooltip.span class, i.e. at the start of the tooltip text.  It doesn't
1653     * actually change the DOM.  Actually it won't be inserted at all: we will take the position of this 
1654     * tooltip content out of the normal document flow anyway by using absolute positioning.
1655     * And that's because we want it hover over the tooltip.
1656     */
1657    span.tooltip:before
1658    {  
1659        /* Pull the tooltip text from the parent HTML span data-text value. */
1660        content:    attr( data-text ) ;
1661
1662        text-align: center ;
1663        color:      white ;
1664        background: darkblue ;
1665     
1666        /* Wide enough for a reasonable chunk of text when hovering. */
1667        width:      10rem ;
1668
1669        /* Take us out of the normal document flow.  The position is relative to the initial containing block,
1670         * which is the edge of the padding box of the nearest ancestor element which has a relative position,
1671         * in our case, span.tooltip.  So long story short, the tooltip contents will hover in a box whose upper
1672         * left corner is the upper left corner padding of the tooltip text.  But then we 
1673         * position using top and transform.
1674         */
1675        position: absolute ;
1676
1677        left: 50% ;                     /* Place left edge of box 1/2 to the right in the tooltip. */
1678        transform: translateX( -50% ) ; /* Then shift it 1/2 way up to center horizontally. */
1679        top: -1rem ;  /* Center vertically. */
1680
1681        padding:       1rem ;
1682        border-radius: 1.5rem ;
1683
1684        /* Tooltip contents is stacked on top of all other display elements. */
1685        z-index: 99 ;
1686
1687        /* Won't show up until we hover. */
1688        opacity: 0 ;

1693    }
